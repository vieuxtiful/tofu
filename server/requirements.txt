## ── real text shaping (knead layer) ────────────────────────────────────
## Pillow's BASIC layout engine applies no GPOS (no kerning, no mark
## attachment), no GSUB (no ligatures, no Indic conjuncts) and no
## reordering. Raqm is the in-Pillow fix but the Windows wheel is compiled
## WITHOUT it -- _imagingft.HAVE_RAQM is False, a compile-time constant no
## DLL can flip -- so ToFU shapes text itself. Both ship Windows wheels and
## need no C toolchain on the target machine.
uharfbuzz==0.55.0           # HarfBuzz: codepoints → positioned glyph IDs
freetype-py==2.5.1          # FreeType: glyph ID → coverage bitmap
xlrd==2.0.2                 # glossary .xls parsing
openpyxl==3.1.5             # glossary .xlsx parsing
scikit-image==0.26.0        # imaging.text_mask local thresholding (Sauvola)
scipy==1.18.0               # scikit-image backbone; pinned to stop drift

## ── one cv2, and only one ──────────────────────────────────────────────
## install EXACTLY ONE opencv distribution. two of them unpack into the
## same site-packages/cv2/ directory, so the winner is whichever wrote
## last — the manifest stops describing what actually imports.
##
## this bit us for real: opencv-contrib-python 4.10.0.84 was shadowing a
## pinned headless 5.0.0.93, so cv2.__version__ reported 4.10 and
## scene.ClassicalCVBackend's contour rescue silently missed the banner
## on images/japan-street.jpeg — test_scene_regions.py::TestContourRescuePass
## ::test_dense_scene_recovers_a_real_quad_for_the_banner failed on main.
## removing the duplicate fixed the test with no code change.
##
## headless is correct here: the server never opens a GUI window, and the
## contrib modules (ximgproc, aruco, xfeatures2d, ...) are used NOWHERE in
## src/, server/, or scripts/ — verified by grep. do not "upgrade" to
## opencv-contrib-python to get a module; add it deliberately or not at all.
opencv-python-headless==5.0.0.93

## ── separate table settings: the isolated interpreters ─────────────────
## paddleocr is deliberately ABSENT from this file. paddlepaddle
## force-replaces the app venv's numpy/opencv on install (measured, CP-1),
## so PP-OCRv5 runs OUT-OF-PROCESS under .venv-paddle via
## scripts/paddle_worker.py. EasyOCR (Latin) and PaddleOCR (CJK) are both
## load-bearing — the isolation is what lets them coexist, not a downgrade.
##   .venv-paddle   -> server/requirements-paddle.txt
##   .venv-inpaint  -> server/requirements-inpaint.txt
